# CUORE companion service.
#
# Install into the repo venv, which already carries the mes library's own
# dependencies:
#     .venv/Scripts/python.exe -m pip install -r cuore/requirements.txt
#
# Plain `uvicorn`, not `uvicorn[standard]`: the extras pull httptools and
# uvloop, which are the usual source of build failures on Windows and give
# nothing this workload needs. `websockets` is pinned in explicitly instead --
# it is the one extra the live stream will use at P3.

fastapi>=0.115
uvicorn>=0.30
websockets>=12
jinja2>=3.1
pydantic>=2.7
pydantic-settings>=2.2
